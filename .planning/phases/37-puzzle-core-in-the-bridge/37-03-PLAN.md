#[derive(Debug, Clone, Copy, PartialEq)]
pub(crate) struct GoalRect { min: Vec2, max: Vec2 }
impl GoalRect {
    pub(crate) const fn new(min: Vec2, max: Vec2) -> Self;
    pub(crate) const fn min(self) -> Vec2;
    pub(crate) const fn max(self) -> Vec2;
    /// Finite corners with min strictly below max on both axes.
    pub(crate) fn is_valid(self) -> bool;
    /// Strict containment: a centre on any edge is outside; NaN is outside.
    pub(crate) fn contains_strictly(self, point: Vec2) -> bool;
}
pub(crate) fn count_in_goal<G: PartialEq>(positions: &[Vec2], groups: &[Option<G>], counted: &[G], interior: GoalRect) -> u32;
pub(crate) fn fill_ratio(count: u32, full_count: u32) -> f64;          // min(count/full, 1.0); 0.0 when full_count == 0
pub(crate) const fn fill_margin_count(full_count: u32) -> u32;         // full_count.div_ceil(20)

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub(crate) enum Outcome { Running, Won { step: u32 }, Lost { step: u32 } }
impl Outcome {
    pub(crate) const fn code(self) -> u8;              // 0 running, 1 won, 2 lost (D-17)
    pub(crate) const fn maybe_step(self) -> Option<u32>; // None while running
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub(crate) struct GoalTracker { target: u32, settle_steps: u32, step_limit: u32, held: u32, outcome: Outcome }
impl GoalTracker {
    /// None unless 1 <= target, 1 <= settle_steps <= step_limit.
    pub(crate) const fn new(target: u32, settle_steps: u32, step_limit: u32) -> Option<Self>;
    pub(crate) fn observe(&mut self, count: u32, step: u32) -> Outcome;
    pub(crate) const fn outcome(&self) -> Outcome;
}
