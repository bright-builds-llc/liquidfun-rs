```ts
export function unprojectPoint(camera: Camera, point: Point): Point {
  return {
    x: WORLD_BOUNDS.minX + (point.x - camera.offsetX) / camera.scale,
    y: WORLD_BOUNDS.maxY - (point.y - camera.offsetY) / camera.scale,
  };
}
```

```ts
export type PointerKind = "down" | "move" | "up" | "cancel";

export function parsePointerKind(raw: string): PointerKind | undefined;

export type ClientRectLike = {
  readonly left: number;
  readonly top: number;
  readonly width: number;
  readonly height: number;
};

export function cssPointFromClient(
  clientX: number,
  clientY: number,
  rect: ClientRectLike,
): Point | undefined;

export type GestureState = {
  readonly maybePointerId: number | undefined;
};

export type PointerEventName =
  | "pointerdown"
  | "pointermove"
  | "pointerup"
  | "pointercancel"
  | "lostpointercapture";

export type GestureDecision = {
  readonly next: GestureState;
  readonly maybeKind: PointerKind | undefined;
  readonly shouldCapture: boolean;
  readonly shouldRelease: boolean;
};

export function emptyGestureState(): GestureState;
export function reducePointerEvent(
  state: GestureState,
  event: { readonly name: PointerEventName; readonly pointerId: number },
): GestureDecision;
```

`cssPointFromClient` computes `{ x: clientX - rect.left, y: clientY - rect.top }` and returns `undefined` when any input or result is not finite, or when `rect.width <= 0` or `rect.height <= 0`. It must not multiply by `rect.width`, device pixel ratio, or a backing-store size.

`reducePointerEvent` rules (D-05):
1. `pointerdown` → `maybeKind: "down"`, `shouldCapture: true`, store `pointerId`.
1. `pointermove` → `maybeKind: "move"` only when `state.maybePointerId === pointerId`; otherwise ignore.
1. `pointerup` for the stored id → `maybeKind: "up"`, clear id, `shouldRelease: true`.
1. `pointercancel` or `lostpointercapture` for the stored id → `maybeKind: "cancel"`, clear id, `shouldRelease: true`.
1. Events for a different id, or cancel/lost when no id is stored, return `maybeKind: undefined` and leave state unchanged.
1. Do not read `event.isTrusted`.
