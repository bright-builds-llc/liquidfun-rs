```ts
// web/src/input/canvas-pointer.ts
export type CanvasPointerHandlers = {
  cancel(): void;
  detach(): void;
};

export function attachCanvasPointer(options: {
  readonly canvas: HTMLCanvasElement;
  readonly maybeCamera: () => Camera | undefined;
  readonly send: (kind: PointerKind, worldX: number, worldY: number) => void;
}): CanvasPointerHandlers;
```

Adapter rules (D-05, D-06, D-07):
1. Listen for `pointerdown`, `pointermove`, `pointerup`, `pointercancel`, `lostpointercapture` only. No `mousedown` / `touchstart`.
1. On each event, `rect = canvas.getBoundingClientRect()`, `css = cssPointFromClient(event.clientX, event.clientY, rect)`, `camera = maybeCamera()`, then `world = unprojectPoint(camera, css)` when all are defined.
1. Feed `{ name: event.type, pointerId: event.pointerId }` to `reducePointerEvent`. If `shouldCapture`, `canvas.setPointerCapture(pointerId)` then `event.preventDefault()` (canvas only). If `shouldRelease` and `canvas.hasPointerCapture(pointerId)`, `releasePointerCapture`.
1. Call `send(kind, world.x, world.y)` when `maybeKind` is defined and world is finite. If CSS/camera/world is missing, still honor capture/release/cancel so leftover capture cannot stick.
1. `cancel()`: if `maybePointerId` is set, `releasePointerCapture` when held, `send("cancel", lastWorldX, lastWorldY)` or `send("cancel", 0, 0)` if no last world sample, then reset reducer state.
1. `detach()` calls `cancel()` then removes the five listeners.
1. Do not check `event.isTrusted`.
