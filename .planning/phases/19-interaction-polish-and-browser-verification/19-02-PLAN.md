```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub(crate) enum PointerKind {
    Down,
    Move,
    Up,
    Cancel,
}

pub(crate) fn parse_pointer_kind(raw: &str) -> Result<PointerKind, SessionError> {
    match raw {
        "down" => Ok(PointerKind::Down),
        "move" => Ok(PointerKind::Move),
        "up" => Ok(PointerKind::Up),
        "cancel" => Ok(PointerKind::Cancel),
        _ => Err(SessionError::UnknownControl),
    }
}

pub(crate) trait SceneHooks {
    // existing methods unchanged
    fn apply_pointer(
        &mut self,
        world: &mut World,
        system: ParticleSystemId,
        kind: PointerKind,
        world_x: f32,
        world_y: f32,
    ) -> Result<(), SessionError>;
}
```

```rust
// SessionError — add this variant and message
InvalidPointer, // "Rust/WASM pointer coordinates must be finite"

// SessionCore
pub(crate) fn apply_pointer(&mut self, kind: &str, world_x: f32, world_y: f32) -> Result<(), SessionError> {
    let parsed = parse_pointer_kind(kind)?;
    if !world_x.is_finite() || !world_y.is_finite() {
        return Err(SessionError::InvalidPointer);
    }
    self.hooks.apply_pointer(&mut self.world, self.particle_system, parsed, world_x, world_y)
}
```

```rust
#[wasm_bindgen(js_name = pointerAction)]
pub fn pointer_action(&mut self, kind: String, world_x: f32, world_y: f32) -> Result<(), JsError> {
    self.core.apply_pointer(&kind, world_x, world_y).map_err(js_error)
}
```
