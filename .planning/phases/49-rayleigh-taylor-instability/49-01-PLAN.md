struct RayleighTaylorHooks {
    heavy_group: ParticleGroupId,
    ratio: WeightRatio,
    scale: ForceScale,
    steps_advanced: u16, // saturating; every rebuild starts at 0
    settle_steps: u16,
    ramp_steps: u16,

fn ramp_factor(step: u16, settle_steps: u16, ramp_steps: u16) -> f32 {
    let Some(into_ramp) = step.checked_sub(settle_steps) else { return 0.0; };
    (f32::from(into_ramp.saturating_add(1)) / f32::from(ramp_steps)).min(1.0)
}
