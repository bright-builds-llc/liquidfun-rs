```typescript
export const PROGRESS_VERSION = 1;
/** Rust `levelInfo().levelHash()` form (D-17). */
export const LEVEL_HASH_PATTERN = /^[0-9a-f]{16}$/;

/** Last placements for a level; `tools` stay raw until restore validates them (D-18). */
export type SavedPlacements = { readonly levelHash: string; readonly tools: readonly unknown[] };
/** Stars plus optional saved placements for one known level. */
export type LevelProgress = LevelStars & { readonly maybePlacements: SavedPlacements | undefined };
/** Parsed v1 document. Every record is null-prototype. */
export type ProgressDocument = {
  readonly levels: Readonly<Partial<Record<PuzzleId, LevelProgress>>>;
  /** Raw JSON of level ids this build does not know, written back verbatim (D-13). */
  readonly unknownLevels: Readonly<Record<string, unknown>>;
  /** Raw top-level keys other than `version` and `levels`, written back verbatim. */
  readonly extra: Readonly<Record<string, unknown>>;
};
export type UnusableReason = "json" | "shape" | "future-version";
export type ParsedProgress =
  | { readonly kind: "ok"; readonly progress: ProgressDocument }
  | { readonly kind: "unusable"; readonly reason: UnusableReason };

export const EMPTY_PROGRESS: ProgressDocument;
export function parseProgress(maybeRaw: string | null): ParsedProgress;
export function serializeProgress(progress: ProgressDocument): string;
export function mergeProgress(stored: ProgressDocument, local: ProgressDocument): ProgressDocument;
export function withStars(progress: ProgressDocument, id: PuzzleId, stars: LevelStars): ProgressDocument;
export function withPlacements(progress: ProgressDocument, id: PuzzleId, maybeSaved: SavedPlacements | undefined): ProgressDocument;
/** Persisted best for a level, or undefined before its first win (matches StarStore.get). */
export function maybeLevelStars(progress: ProgressDocument, id: PuzzleId): LevelStars | undefined;
```
Implementation rules:
1. `parseProgress`: `null` → ok + `EMPTY_PROGRESS`. `JSON.parse` inside `try`/`catch` → `json` (the catch returns a typed result; it does not swallow). Not a plain record (`typeof === "object"`, not null, not array) → `shape`. `version` is a number, an integer and `> PROGRESS_VERSION` → `future-version`; otherwise `version !== PROGRESS_VERSION` → `shape`. `levels` not a plain record → `shape`. Iterate `Object.entries(levels)` (own keys only, safe for a JSON `"__proto__"` key). Known ids (a `ReadonlySet<string>` built from `PUZZLE_IDS`) go through `parseLevel`; any `shape` failure there makes the whole document `shape`. Other keys are stored raw in a null-prototype `unknownLevels`. Top-level keys other than `version` and `levels` go raw into a null-prototype `extra`.
2. `parseLevel(value)`: plain record; `win`, `brim`, `thrifty` are booleans; `bestPeakHeldPct` is a finite number in `[0, 100]`; `(brim || thrifty) && !win` → shape. 3. `serializeProgress`: build `levels` with `Object.create(null)`; known ids in `PUZZLE_IDS` order (`{win, brim, thrifty, bestPeakHeldPct}` plus `placements: {levelHash, tools}` when present), then `unknownLevels` entries in insertion order; top object `{version: 1, levels}` then `extra` entries (also null-prototype). `JSON.stringify` the result.
