fn prepare_create(&self, input: ParticleInput, diagnostic_id: u64, free_slots: usize) -> Result<CreateCandidate, ParticleStorageError> {
    // capacity check → validate_appended_group(input.maybe_group)? → identity_slot_candidate(free_slots)? → id
    let mut groups = self.groups.clone(); groups.push(input.maybe_group);                         // O(n) clone
    let group_records = rebuild_group_records_for_system(&self.group_records, &groups, self.system)?; // O(n) membership_ranges + validate_groups

    Ok(CreateCandidate { input, diagnostic_id, id, local_slot, generation, append_identity, dense, group_records, solver_state })
}
